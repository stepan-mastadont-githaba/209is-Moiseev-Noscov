git init -- инициирует новый репозиторий
git clone <url> -- клонирует удаленный репозиторий по url
git add <file> -- добавляет файл для последующего коммита. git add . для всех изменений
git commit -m "*сообщение*" -- коммитит указанные изменения
git status -- текущее состояние рабочей директории
git log -- выводит историю коммитов текущей ветки
git diff -- показывает изменения в файлах, которые еще не добавлены
git branch -- список локальных веток, git branch <имя> чтобы создать новую
git checkout <ветка> -- переключает на указанную ветку
git merge <ветка> -- сливает указанную ветку с текущей веткой
git pull <remote> <branch> -- отправляет на удаленный репозиторий
git remote -v -- показывает список репозиториев
git reset <commit> -- возвращает коммит к исходному состоянию
git revert <commit> -- создает новый коммит, который отменяет изменения
git stash -- сохраняет изменения во временное хранилище
git stash pop -- возвращает изменения из хранилища
git tag <имя> -- создает тег для коммита
git rebase <ветка> применяет коммит для другой ветки


Понятие репозитория и структура проекта
репозиторий - это хранилище кода проекта, включающее:
    1. все файлы и папки проекта
    2. историю изменений (коммиты)
    3. информацию о ветках, тегах и настройках

виды репозиториев:
    1. папка git (скрыта)
    2. удаленный репозиторий
    3. github gitlab

my_project/
│
├── .gitignore              # Файл для игнорирования определенных файлов и папок в Git
├── README.md               # Файл описания проекта (инструкции, как использовать проект)
├── LICENSE                 # Файл лицензии (например, MIT, Apache, GPL и др.)
│
├── src/                    # Исходный код проекта
│   ├── __init__.py         # Инициализационный файл для пакета
│   ├── main.py             # Главный файл приложения
│   └── module1.py          # Дополнительные модули
│
├── tests/                  # Тесты проекта
│   ├── __init__.py         # Инициализационный файл для пакета
│   ├── test_module1.py     # Тесты для module1
│   └── test_module2.py     # Тесты для других модулей
│
├── requirements.txt        # Список зависимостей (для Python-проектов)
├── setup.py                # Скрипт для установки пакета (для Python-проектов)
│
├── docs/                   # Документация проекта
│   ├── index.md            # Главный файл документации
│   └── usage.md            # Инструкции по использованию
│
└── examples/               # Примеры использования вашего проекта
    ├── example1.py         # Пример 1
    └── example2.py         # Пример 2
